--查询数据库中是否包含某个语句
SELECT 
    o.name AS ObjectName,
    o.type_desc AS ObjectType,
    m.definition AS ObjectDefinition
FROM sys.sql_modules m
JOIN sys.objects o ON m.object_id = o.object_id
WHERE m.definition LIKE '%update zj_YaoPinTaiZhangNEW%'  -- 替换为你的目标语句
AND o.type IN ('P', 'TR', 'FN', 'IF', 'TF');  -- 指定对象类型



-- 改进版：查找高IO操作的表（包含更多有用信息）
SELECT TOP 10 
    OBJECT_NAME(s.object_id) AS table_name,
    SUM(user_seeks + user_scans + user_lookups) AS total_operations,
    SUM(user_seeks) AS index_seeks,           -- 高效查找次数
    SUM(user_scans) AS index_scans,           -- 索引扫描次数  
    SUM(user_lookups) AS key_lookups,         -- 书签查找次数
    SUM(user_updates) AS update_operations     -- 更新操作次数
FROM sys.dm_db_index_usage_stats s
WHERE database_id = DB_ID('FLB')
    AND OBJECT_NAME(s.object_id) IS NOT NULL  -- 排除已删除的表
GROUP BY OBJECT_NAME(s.object_id)
ORDER BY total_operations DESC;               -- 按总操作次数排序


-- 1. 查看当前活跃的数据库查询
SELECT 
    session_id, blocking_session_id, wait_type, wait_time, 
    total_elapsed_time, cpu_time, logical_reads,
    SUBSTRING(text, (statement_start_offset/2)+1, 
        ((CASE statement_end_offset WHEN -1 THEN DATALENGTH(text)
          ELSE statement_end_offset END - statement_start_offset)/2)+1) AS query_text
FROM sys.dm_exec_requests 
CROSS APPLY sys.dm_exec_sql_text(sql_handle)
WHERE session_id > 50 AND session_id <> @@SPID
ORDER BY logical_reads DESC;
